\section{Proofs and further results for Sections~\ref{sec:identification}--\ref{sec:measurement}}\label{app:proofs-design}

Throughout, $kTS$ is the number of recorded positions of one puppet, and for a puppet $u$ we write $\mathrm{TP}_u$, $\mathrm{FP}_u$, $\mathrm{FN}_u$ and $\mathrm{TN}_u$ for the numbers of its recorded positions that are true positives, false positives, false negatives and true negatives, and $\mathrm{PP}_u=\mathrm{TP}_u+\mathrm{FP}_u$ and $\mathrm{PN}_u=\mathrm{FN}_u+\mathrm{TN}_u$ for the numbers of its flagged (predicted positive) and unflagged positions. Thus $kTS\,\bar E_u=\mathrm{TP}_u+\mathrm{FN}_u$ and $kTS\,\bar Q_u=\mathrm{PP}_u$. All statements are conditional on the environment of the window (\cref{sec:id-design}). We use a superpopulation formulation: the vectors of potential records $\{\mathcal R_u(z):z\in\mathcal D_A\times\{\mathrm{alg},\mathrm{ref}\}\}$ of distinct puppets are independent, and the assignment $((D_u,\Pi_u))_u$ is independent of them.

\subsection{Two counting lemmas}\label{app:pf-lemmas}

\begin{lemma}[Counting identity]\label{lem:id-count}
For every cell, $q_\pi(d)=\Se_\pi(d)\mu_\pi(d)+\{1-\Sp_\pi(d)\}\{1-\mu_\pi(d)\}=\{1-\Sp_\pi(d)\}+J_\pi(d)\mu_\pi(d)$, and $q_\pi(d)\PPV_\pi(d)=\Se_\pi(d)\mu_\pi(d)$, where $\PPV_\pi(d)=\E\mathrm{TP}_u/\E\mathrm{PP}_u$ is the exposure-weighted positive predictive value.
\end{lemma}

\begin{proof}
Take expectations in $kTS\,\bar Q_u=\mathrm{TP}_u+\mathrm{FP}_u$. By definition \eqref{eq:id-sesp}, $\E\mathrm{TP}_u=\Se_\pi(d)\,\E(\mathrm{TP}_u+\mathrm{FN}_u)=\Se_\pi(d)\,kTS\,\mu_\pi(d)$, and $\E\mathrm{FP}_u=\E(\mathrm{FP}_u+\mathrm{TN}_u)-\E\mathrm{TN}_u=\{1-\Sp_\pi(d)\}\,kTS\,\{1-\mu_\pi(d)\}$. Dividing by $kTS$ gives the first identity, and rearranging gives the second. Finally, $q_\pi(d)\PPV_\pi(d)=\E\mathrm{TP}_u/(kTS)=\Se_\pi(d)\mu_\pi(d)$. (If $\mu_\pi(d)=0$ or $1$, the undefined rate can be given any value and the identities still hold.)
\end{proof}

\begin{lemma}[Confirmed-flagged exposure]\label{lem:seq-unbiased}
Let $Y_u=\bar Q_u\widehat P_u$ as in \cref{sec:meas-sequential}, with $Y_u=0$ if the puppet has no flagged position. Then $\E(Y_u\mid\mathcal R_u)=\mathrm{TP}_u/(kTS)$, and $\E Y_u=\Se_\pi(d)\mu_\pi(d)$ for a puppet of cell $(d,\pi)$.
\end{lemma}

\begin{proof}
If $\mathrm{PP}_u=0$ then $Y_u=0=\mathrm{TP}_u/(kTS)$. If $0<\mathrm{PP}_u\le m$, all flagged positions are coded, so $\widehat P_u=\mathrm{TP}_u/\mathrm{PP}_u$. If $\mathrm{PP}_u>m$, the number of confirmed positives among $m$ positions drawn without replacement is hypergeometric with mean $m\mathrm{TP}_u/\mathrm{PP}_u$. In every case $\E(\widehat P_u\mid\mathcal R_u)=\mathrm{TP}_u/\mathrm{PP}_u$, and since $\bar Q_u=\mathrm{PP}_u/(kTS)$ is a function of the record, $\E(Y_u\mid\mathcal R_u)=\mathrm{TP}_u/(kTS)$. Taking expectations and using \eqref{eq:id-sesp} gives $\E Y_u=\Se_\pi(d)\mu_\pi(d)$.
\end{proof}

\subsection{Proof of \texorpdfstring{\cref{thm:identify}}{the identification theorem}}\label{app:pf-identify}

\emph{(a) Identification.} By \cref{asm:id-nointerf}, the observed record of puppet $u$ is $\mathcal R_u(D_u,\Pi_u)$. Because the assignment is independent of the potential records, $\Pr\{\mathcal R_u(D_u,\Pi_u)\in\cdot\mid(D_u,\Pi_u)=(d,\pi)\}=\Pr\{\mathcal R_u(d,\pi)\in\cdot\}$. By \cref{asm:id-stable}(a), this law $F_{d\pi}$ is the same for every puppet. Hence $q_\pi(d)=\E\{\bar Q_u\mid(D_u,\Pi_u)=(d,\pi)\}$ is the mean of $\bar Q$ under $F_{d\pi}$, a functional of the law of the observed data. By \cref{asm:id-script}, $\bar E_u$ under $F_{d\pi}$ is the session average of draws of $E_u(\pi;d)$, so its mean is $\mu_\pi(d)$. \Cref{lem:id-count} with \cref{asm:id-nondiff} gives $q_\pi(d)=\{1-\Sp(d)\}+J(d)\mu_\pi(d)$, and since $J(d)>0$, $\mu_\pi(d)=\{q_\pi(d)+\Sp(d)-1\}/J(d)$. The expressions for $\mathrm{AA}(d)$ and $\mathrm{AR}(d)$ follow by subtraction and division within persona $d$, in which $\Sp(d)-1$ cancels from the difference and $J(d)$ from the ratio. A linear combination $\sum c_{d\pi}\mu_\pi(d)$ equals $\sum c_{d\pi}\{q_\pi(d)+\Sp(d)-1\}/J(d)$. The three terms of the persona-contrast decomposition are such combinations. For the demand contrast, $\mu_{\mathrm{ref}}(d)-\mu_{\mathrm{ref}}(d_0)=\{q_{\mathrm{ref}}(d)+\Sp(d)-1\}/J(d)-\{q_{\mathrm{ref}}(d_0)+\Sp(d_0)-1\}/J(d_0)$. Neither $\Sp$ nor $J$ cancels unless both rates coincide across the two personas. Correcting with common rates when they differ gives the wrong contrast, by \cref{thm:misclass}(c) applied to personas instead of arms.

\emph{(b) Asymptotic normality.} Conditionally on the assignment, the observed records are independent, and those of the $n_{d\pi}$ puppets of cell $(d,\pi)$ are identically distributed with law $F_{d\pi}$. Since $\bar Q_u\in[0,1]$, the variance $\sigma^2_{d\pi}$ is finite, and the central limit theorem gives $\sqrt{n_{d\pi}}(\widehat q_{d\pi}-q_\pi(d))\Rightarrow\mathrm N(0,\sigma^2_{d\pi})$. With $n_{d\pi}/n\to p_{d\pi}$, $\sqrt n(\widehat q_{d\pi}-q_\pi(d))\Rightarrow\mathrm N(0,\sigma^2_{d\pi}/p_{d\pi})$. The cell means are independent given the assignment, and since the limit does not depend on the assignment, they are jointly asymptotically normal with a diagonal covariance matrix. The map $(q_a,q_r)\mapsto(q_a-q_r)/J$ is linear, which gives the law of $\widehat{\mathrm{AA}}(d)$. The map $(q_a,q_r)\mapsto\log(q_a+\Sp-1)-\log(q_r+\Sp-1)$ is differentiable at the truth when $\mu_{\mathrm{alg}}(d),\mu_{\mathrm{ref}}(d)>0$, with gradient $\{1/(J\mu_{\mathrm{alg}}),-1/(J\mu_{\mathrm{ref}})\}$, so the delta method \citep[Ch.~3]{vandervaart1998} gives the law of $\log\widehat{\mathrm{AR}}(d)$. Finally, $\sigma^2_{d\pi}=\Var(S^{-1}\sum_sQ_{us})=S^{-2}\sum_{s,s'}\Cov(Q_{us},Q_{us'})$. If the sessions are exchangeable, with common mean $q$, variance $\sigma^2_{\mathrm s}$ and pairwise correlation $\gamma$, this equals $S^{-2}\{S\sigma^2_{\mathrm s}+S(S-1)\gamma\sigma^2_{\mathrm s}\}=\sigma^2_{\mathrm s}\{1+(S-1)\gamma\}/S$, so $\mathrm{deff}=1+(S-1)\gamma$.

\emph{(c) Variance estimation.} In the saturated regression of $Q_{us}$ on cell indicators, the coefficient of cell $(d,\pi)$ is the mean of its $n_{d\pi}S$ sessions, which equals $\widehat q_{d\pi}$ because every puppet has $S$ sessions. With $S$ sessions for every puppet, its bread is $(n_{d\pi}S)^{-1}$ and its residuals are $e_{us}=Q_{us}-\widehat q_{d\pi}$. The cluster-robust meat is $\sum_u(\sum_se_{us})^2=S^2\sum_u(\bar Q_u-\widehat q_{d\pi})^2$. The sandwich is therefore $n_{d\pi}^{-2}\sum_u(\bar Q_u-\widehat q_{d\pi})^2$, and multiplying by $n_{d\pi}/(n_{d\pi}-1)$ gives $s^2_{d\pi}/n_{d\pi}$. By the strong law of large numbers, $s^2_{d\pi}\to\sigma^2_{d\pi}$ almost surely, and Slutsky's lemma gives asymptotic coverage $1-\alpha$ for the Wald intervals. The estimator that treats sessions as independent is $\widehat\sigma^2_{\mathrm s,d\pi}/(n_{d\pi}S)$, where $\widehat\sigma^2_{\mathrm s,d\pi}$ is the sample variance of the $n_{d\pi}S$ values $Q_{us}$. By the strong law applied to the puppet-level sums $\sum_s(Q_{us}-q)^2$, $\widehat\sigma^2_{\mathrm s,d\pi}\to S^{-1}\sum_s\E(Q_{us}-q)^2=\sigma^2_{\mathrm s,d\pi}$. Hence $n_{d\pi}$ times this estimator converges to $\sigma^2_{\mathrm s,d\pi}/S=\sigma^2_{d\pi}/\mathrm{deff}_{d\pi}$, which differs from $\sigma^2_{d\pi}$ unless $\mathrm{deff}_{d\pi}=1$ and is smaller when $\mathrm{deff}_{d\pi}>1$.

\emph{(d) Blocked design.} Under \cref{asm:id-stable}(b), randomisation within waves makes the observed records of the puppets of wave $w$ and cell $(d,\pi)$ independent draws from $F_{wd\pi}$, and records of different waves independent. Part (a), applied within wave $w$ with the wave's error rates (equal across arms), gives $\mu_{w\pi}(d)=\{q_{w\pi}(d)+\Sp_w-1\}/J_w$. Hence $\widehat\mu_{wd\pi}=(\widehat q_{wd\pi}+\Sp_w-1)/J_w$ is unbiased for $\mu_{w\pi}(d)$, and $\sum_w\omega_w\widehat\mu_{wd\pi}$ is unbiased for $\bar\mu_\pi(d)$. With $W$ fixed and $n_{wd\pi}/n\to p_{wd\pi}>0$, the $2W$ wave-cell means are independent and asymptotically normal as in (b), so $\sqrt n\{\sum_w\omega_w\widehat\mu_{wd\pi}-\bar\mu_\pi(d)\}\Rightarrow\mathrm N\{0,\sum_w\omega_w^2\sigma^2_{wd\pi}/(J_w^2p_{wd\pi})\}$, independently for the two arms. The difference and the delta method for the log ratio, with gradient $(1/\bar\mu_{\mathrm{alg}},-1/\bar\mu_{\mathrm{ref}})$, give the stated laws. Consistency of the stratified variance estimator follows wave by wave as in (c).
\qed

\subsection{Interference: proofs and further results}\label{app:pf-interference}

\emph{Proof of \cref{prop:interference}(a).} Under \cref{asm:id-bounded}, waves are independent, and within wave $w$, given the load $\ell_w$, the puppets of cell $(d,\pi)$ are independent draws from a law with mean exposure $m_{d\pi}(\ell_w)$ and arm-independent error rates. This is \cref{asm:id-stable}(b) with $F_{wd\pi}$ the law at load $\ell_w$. By \cref{thm:identify}(d), each corrected wave-cell mean is unbiased for $m_{d\pi}(\ell_w)$, so $\E\widehat{\mathrm{AA}}(d)=\bar m_{\mathrm{alg}}(d)-\bar m_{\mathrm{ref}}(d)$ with $\bar m_\pi(d)=\sum_w\omega_wm_{d\pi}(\ell_w)$. Writing $\mathrm{AA}(\ell)=m_{d,\mathrm{alg}}(\ell)-m_{d,\mathrm{ref}}(\ell)$, so that $\mathrm{AA}(0)=\mathrm{AA}^0(d)$,
\[
  \bigl|\E\widehat{\mathrm{AA}}(d)-\mathrm{AA}^0(d)\bigr|=\Bigl|\sum_w\omega_w\{\mathrm{AA}(\ell_w)-\mathrm{AA}(0)\}\Bigr|\le\sum_w\omega_w\xi_\Delta\ell_w=\xi_\Delta\bar\ell .
\]
By the triangle inequality, $|\mathrm{AA}(\ell)-\mathrm{AA}(0)|\le|m_{d,\mathrm{alg}}(\ell)-m_{d,\mathrm{alg}}(0)|+|m_{d,\mathrm{ref}}(\ell)-m_{d,\mathrm{ref}}(0)|\le(\xi_{\mathrm{alg}}+\xi_{\mathrm{ref}})\ell$, so the smallest admissible $\xi_\Delta$ is no larger than $\xi_{\mathrm{alg}}+\xi_{\mathrm{ref}}$ (an arbitrary admissible $\xi_\Delta$ need not be). For the ratio, $|\bar m_\pi(d)-\mu_\pi(d)|\le\xi_\pi\bar\ell$ in the same way. If $\bar m_\pi(d)>\xi_\pi\bar\ell$, then $\min\{\bar m_\pi(d),\mu_\pi(d)\}\ge\bar m_\pi(d)-\xi_\pi\bar\ell>0$, and the mean value theorem gives $|\log\bar m_\pi(d)-\log\mu_\pi(d)|\le|\bar m_\pi(d)-\mu_\pi(d)|/\min\{\bar m_\pi(d),\mu_\pi(d)\}\le\xi_\pi\bar\ell/\{\bar m_\pi(d)-\xi_\pi\bar\ell\}$. Adding the two arms bounds $|\log\{\bar m_{\mathrm{alg}}(d)/\bar m_{\mathrm{ref}}(d)\}-\log\mathrm{AR}^0(d)|$. For attainment, take $m_{d,\mathrm{alg}}(\ell)=\mu_{\mathrm{alg}}(d)+\xi_\Delta\ell$ and $m_{d,\mathrm{ref}}(\ell)=\mu_{\mathrm{ref}}(d)$, which satisfies \cref{asm:id-bounded} with $\xi_{\mathrm{alg}}=\xi_\Delta$ and $\xi_{\mathrm{ref}}=0$ and has bias exactly $\xi_\Delta\bar\ell$. Taking instead $m_{d,\mathrm{alg}}(\ell)=\mu_{\mathrm{alg}}(d)+\xi_{\mathrm{alg}}\ell$ and $m_{d,\mathrm{ref}}(\ell)=\mu_{\mathrm{ref}}(d)-\xi_{\mathrm{ref}}\ell$, with $\xi_{\mathrm{ref}}\ell_{\max}\le\mu_{\mathrm{ref}}(d)$, gives $\xi_\Delta=\xi_{\mathrm{alg}}+\xi_{\mathrm{ref}}$ and bias $(\xi_{\mathrm{alg}}+\xi_{\mathrm{ref}})\bar\ell$.

\emph{Proof of \cref{prop:interference}(d).} By (a), $\mathrm{AA}^0(d)\in[\E\widehat{\mathrm{AA}}-\xi_\Delta\bar\ell,\E\widehat{\mathrm{AA}}+\xi_\Delta\bar\ell]$. On the event $|\widehat{\mathrm{AA}}-\E\widehat{\mathrm{AA}}|\le z_{1-\alpha/2}\widehat{\mathrm{se}}$, whose probability is $1-\alpha$ to the accuracy of the normal approximation of \cref{thm:identify}, the interval $\widehat{\mathrm{AA}}\pm(\xi_\Delta\bar\ell+z_{1-\alpha/2}\widehat{\mathrm{se}})$ contains $\mathrm{AA}^0(d)$. It excludes zero if and only if $|\widehat{\mathrm{AA}}|>\xi_\Delta\bar\ell+z_{1-\alpha/2}\widehat{\mathrm{se}}$, that is, if and only if $\xi_\Delta<\xi^\ast$. Parts (b) and (c) are \cref{prop:int-direction,prop:int-saturation} below.
\qed

\begin{proposition}[Direction at a fixed state]\label{prop:int-direction}
Suppose a policy fills each recorded position independently, choosing item $c$ with probability proportional to $f_c\,p_c^{\,\theta}$, where $f_c\ge0$ is a relevance factor fixed by the puppet's current state, $p_c$ is the item's popularity and $\theta>0$. Suppose the wave's puppets shift popularity to $p=(1-\ell)p^0+\ell p^{\mathrm P}$, where $p^{\mathrm P}$ is the popularity generated by the puppets' openings. Let $\chi(\ell)$ be the probability that a position shows an item with $L_c=1$.
\begin{enumerate}[(i)]
  \item If $p^{\mathrm P}$ puts no mass on items with $L_c=1$, then $\chi(\ell)$ is non-increasing in $\ell$.
  \item If $\theta=1$, then for every $p^{\mathrm P}$, $\chi(\ell)=(1-\epsilon_\ell)\,\chi(0)+\epsilon_\ell\,\chi^{\mathrm P}$, where $\chi^{\mathrm P}=\sum_{c:L_c=1}f_cp^{\mathrm P}_c/\sum_cf_cp^{\mathrm P}_c$ and $\epsilon_\ell\in[0,1)$ increases from $\epsilon_0=0$. Exposure then rises with load if and only if $\chi^{\mathrm P}>\chi(0)$, which a small share of puppet engagement on false negatives with large $f_c$ can produce.
\end{enumerate}
\end{proposition}

\begin{proof}
Fix the puppet's state, and so the factors $f_c$. Let $\mathcal A_1=\{c:L_c=1\}$, and for a popularity vector $p$ let $s_1(p)=\sum_{c\in\mathcal A_1}f_cp_c^\theta$ and $s(p)=\sum_cf_cp_c^\theta$, assumed positive, so that the probability that a position shows an item with $L_c=1$ is $s_1(p)/s(p)$. Let $0\le\ell<\ell'<1$. Then $p(\ell')=(1-\epsilon)p(\ell)+\epsilon p^{\mathrm P}$ with $\epsilon=(\ell'-\ell)/(1-\ell)\in(0,1]$, as one checks by expanding both sides. For $c\in\mathcal A_1$, $p^{\mathrm P}_c=0$, so $p_c(\ell')=(1-\epsilon)p_c(\ell)$ and $s_1\{p(\ell')\}=(1-\epsilon)^\theta s_1\{p(\ell)\}$. For $c\notin\mathcal A_1$, $p_c(\ell')\ge(1-\epsilon)p_c(\ell)$, and $x\mapsto x^\theta$ is increasing, so $s\{p(\ell')\}-s_1\{p(\ell')\}\ge(1-\epsilon)^\theta[s\{p(\ell)\}-s_1\{p(\ell)\}]$. Hence $s\{p(\ell')\}\ge(1-\epsilon)^\theta s\{p(\ell)\}$ and $s_1\{p(\ell')\}/s\{p(\ell')\}\le s_1\{p(\ell)\}/s\{p(\ell)\}$, which is (i). For (ii), with $\theta=1$, both sums are linear in $p$, so $s_1\{p(\ell)\}=(1-\ell)s_1(p^0)+\ell s_1(p^{\mathrm P})$ and $s\{p(\ell)\}=(1-\ell)s(p^0)+\ell s(p^{\mathrm P})$. Hence
\[
  \chi(\ell)=(1-\epsilon_\ell)\frac{s_1(p^0)}{s(p^0)}+\epsilon_\ell\frac{s_1(p^{\mathrm P})}{s(p^{\mathrm P})},\qquad \epsilon_\ell=\frac{\ell\,s(p^{\mathrm P})}{(1-\ell)s(p^0)+\ell\,s(p^{\mathrm P})},
\]
with $s_1(p^0)/s(p^0)=\chi(0)$ and $s_1(p^{\mathrm P})/s(p^{\mathrm P})=\chi^{\mathrm P}$. The weight $\epsilon_\ell$ is zero at $\ell=0$ and increases in $\ell$, so $\chi(\ell)-\chi(0)=\epsilon_\ell\{\chi^{\mathrm P}-\chi(0)\}$, whose sign is that of $\chi^{\mathrm P}-\chi(0)$. Both statements fix the factors $f_c$, that is, the puppet's state.
\end{proof}

\begin{proposition}[Randomised saturation]\label{prop:int-saturation}
Under the conditions of \cref{prop:interference}, suppose waves are randomised to two loads $\ell_1<\ell_2$, with weights $\omega^{(1)}$ and $\omega^{(2)}$ summing to one, and let $\mathrm{AA}(\ell)=m_{d,\mathrm{alg}}(\ell)-m_{d,\mathrm{ref}}(\ell)$ be twice differentiable on $[0,\ell_2]$ with $|\mathrm{AA}''|\le\zeta$. The extrapolated estimator $\widehat{\mathrm{AA}}^{\mathrm{ex}}=\{\ell_2\widehat{\mathrm{AA}}(\ell_1)-\ell_1\widehat{\mathrm{AA}}(\ell_2)\}/(\ell_2-\ell_1)$, built from the blocked estimators at each load, satisfies
\[
  \bigl|\E\widehat{\mathrm{AA}}^{\mathrm{ex}}-\mathrm{AA}^0(d)\bigr|\le\tfrac12\,\zeta\,\ell_1\ell_2,
\]
and its variance is $\{\ell_2^2v_1+\ell_1^2v_2\}/(\ell_2-\ell_1)^2$, where $v_j$ is the variance of $\widehat{\mathrm{AA}}(\ell_j)$. The slope $\{\widehat{\mathrm{AA}}(\ell_2)-\widehat{\mathrm{AA}}(\ell_1)\}/(\ell_2-\ell_1)$ estimates the average derivative of $\mathrm{AA}$ on $[\ell_1,\ell_2]$, and a test that it is zero is a test of differential interference.
\end{proposition}

\begin{proof}
Let $f=\mathrm{AA}$ on $[0,\ell_2]$ and let $g$ be the linear function with $g(\ell_j)=f(\ell_j)$, $j=1,2$. Then $g(0)=\{\ell_2f(\ell_1)-\ell_1f(\ell_2)\}/(\ell_2-\ell_1)$. The interpolation remainder for twice-differentiable $f$ is standard: for every $x$ there is $\ell^\ast_x$ in the smallest interval containing $x,\ell_1,\ell_2$ with $f(x)-g(x)=\frac12f''(\ell^\ast_x)(x-\ell_1)(x-\ell_2)$. (Proof: for $x\notin\{\ell_1,\ell_2\}$, the function $y\mapsto f(y)-g(y)-\{f(x)-g(x)\}(y-\ell_1)(y-\ell_2)/\{(x-\ell_1)(x-\ell_2)\}$ vanishes at $\ell_1$, $\ell_2$ and $x$, so by Rolle's theorem its second derivative vanishes at some $\ell^\ast_x$.) At $x=0$ this gives $|f(0)-g(0)|\le\frac12\zeta\ell_1\ell_2$. By \cref{thm:identify}(d) applied to the waves at each load, $\E\widehat{\mathrm{AA}}(\ell_j)=f(\ell_j)$, and the extrapolated estimator is linear, so $\E\widehat{\mathrm{AA}}^{\mathrm{ex}}=g(0)$. The two sets of waves are independent, so the variance of the linear combination is $\{\ell_2^2v_1+\ell_1^2v_2\}/(\ell_2-\ell_1)^2$. The slope estimator converges to $\{f(\ell_2)-f(\ell_1)\}/(\ell_2-\ell_1)=(\ell_2-\ell_1)^{-1}\int_{\ell_1}^{\ell_2}f'$, which is zero when $f$ is constant, that is, without differential interference on $[\ell_1,\ell_2]$.
\end{proof}

\subsection{Rotation when cells cannot share a wave}\label{app:rotation}

If capacity does not allow every cell in every wave, cells can be rotated across waves in a balanced incomplete block design, in which every pair of cells shares a wave equally often. Within-wave contrasts then identify every pairwise contrast, and the window average is estimated by combining them. The contrast between $\mathrm{alg}$ and $\mathrm{ref}$ for the same persona should always share a wave, because it is the contrast that drift would otherwise confound.

\subsection{Proof of \texorpdfstring{\cref{thm:misclass}}{the misclassification theorem}, and the variances of corrected estimators}\label{app:pf-misclass}

\begin{proposition}[Variance of corrected estimators]\label{prop:mc-variance}
Under \cref{asm:id-nondiff}, let $(\widehat\Se,\widehat\Sp)$ come from an external design, and let the numbers of puppets $n_a$, $n_r$ and of coded items $n_1$, $n_0$ grow in fixed proportions. Then $\widehat{\mathrm{AA}}$ and $\log\widehat{\mathrm{AR}}$ of \cref{thm:misclass}(b) have asymptotic variances
\begin{align}
  \Var(\widehat{\mathrm{AA}})&\simeq\frac1{J^2}\Bigl\{v_a+v_r+\mathrm{AA}^2\,(v_{\Se}+v_{\Sp})\Bigr\},\label{eq:mc-varAA}\\
  \Var(\log\widehat{\mathrm{AR}})&\simeq\frac1{J^2}\Bigl\{\frac{v_a}{\mu_a^2}+\frac{v_r}{\mu_r^2}+\Bigl(\frac1{\mu_a}-\frac1{\mu_r}\Bigr)^2v_{\Sp}\Bigr\},\label{eq:mc-varAR}
\end{align}
where $v_\pi=\sigma^2_\pi/n_\pi$ is the cluster-robust variance of $\widehat q_\pi$ (\cref{thm:identify}), $v_{\Se}=\Se(1-\Se)/n_1$ and $v_{\Sp}=\Sp(1-\Sp)/n_0$. Replacing every unknown by its estimate gives consistent variance estimators. The sensitivity enters \eqref{eq:mc-varAA} through $J$ and through $v_{\Se}$; in \eqref{eq:mc-varAR} its estimation error does not appear, and it enters only through the common factor $J^{-2}$. With validation stratified by arm and external designs per arm, $\Var(\widehat\mu^{\mathrm{st}}_\pi)\simeq J_\pi^{-2}\{v_\pi+\mu_\pi^2v_{\Se,\pi}+(1-\mu_\pi)^2v_{\Sp,\pi}\}$, and the two arms are independent.
\end{proposition}

\emph{Proof of \cref{thm:misclass}(a).} By \cref{lem:id-count} and \cref{asm:id-nondiff}, $q_\pi=(1-\Sp)+J\mu_\pi$. Since $\mu_a=\mathrm{AR}\,\mu_r$,
\[
  \frac{q_a}{q_r}=\frac{1-\Sp}{q_r}+\frac{J\mu_r}{q_r}\,\mathrm{AR}=w_0+(1-w_0)\,\mathrm{AR},
\]
because $(1-\Sp)+J\mu_r=q_r$. Here $w_0\ge0$, and $w_0<1$ because $J\mu_r>0$. Also $w_0=0$ if and only if $\Sp=1$. The difference is $q_a-q_r=J(\mu_a-\mu_r)$.

\emph{Proof of \cref{thm:misclass}(b) and \cref{prop:mc-variance}.} Let the numbers of puppets and of coded items grow with $n$ in fixed proportions. By \cref{thm:identify}(b), $\widehat q_a$ and $\widehat q_r$ are independent and asymptotically normal with variances $v_a$ and $v_r$. In the external design, $\widehat\Se$ and $\widehat\Sp$ are proportions in two independent binomial samples, hence asymptotically normal with variances $v_{\Se}$ and $v_{\Sp}$, and independent of each other and of the audit. If the numbers $n_1$ and $n_0$ are themselves random, as when a random sample of exposures is coded, the same holds conditionally on them, and the limit does not depend on them. The vector $(\widehat q_a,\widehat q_r,\widehat\Se,\widehat\Sp)$ is therefore jointly asymptotically normal with a diagonal covariance matrix. Let $\mathrm{RG}_\pi(q_\pi,\Se,\Sp)=(q_\pi+\Sp-1)/(\Se+\Sp-1)$ be the Rogan--Gladen map. Its partial derivatives at the truth are
\[
  \frac{\partial\mathrm{RG}_\pi}{\partial q_\pi}=\frac1J,\qquad
  \frac{\partial\mathrm{RG}_\pi}{\partial\Se}=-\frac{q_\pi+\Sp-1}{J^2}=-\frac{\mu_\pi}{J},\qquad
  \frac{\partial\mathrm{RG}_\pi}{\partial\Sp}=\frac{J-(q_\pi+\Sp-1)}{J^2}=\frac{1-\mu_\pi}{J}.
\]
For $\mathrm{AA}=\mathrm{RG}_a-\mathrm{RG}_r$, the gradient with respect to $(q_a,q_r,\Se,\Sp)$ is $J^{-1}(1,-1,-\mathrm{AA},-\mathrm{AA})$, which gives \eqref{eq:mc-varAA}. For $\log\mathrm{AR}=\log(q_a+\Sp-1)-\log(q_r+\Sp-1)$, in which $J$ has cancelled, the gradient is $\{1/(J\mu_a),-1/(J\mu_r),0,1/(J\mu_a)-1/(J\mu_r)\}$, which gives \eqref{eq:mc-varAR}. The delta method gives asymptotic normality, and the continuous mapping theorem gives consistency of the plug-in variance estimators. The stratified variance is the same computation for each arm with its own rates, the arms being independent.

\begin{lemma}[Range of the naive ratio]\label{lem:mc-range}
In the setting of \cref{thm:misclass}(c), if the error rates are only known to lie in $\Se_\pi\in[s_-,s_+]$ and $\Sp_\pi\in[p_-,p_+]$, with $s_-+p_->1$, the set of values of $\mathrm{AR}^{\mathrm{nv}}$ compatible with $(\mu_a,\mu_r)$ is the interval
\[
  \mathrm{AR}\cdot\Bigl[\frac{r_-(\mu_a)}{r_+(\mu_r)},\;\frac{r_+(\mu_a)}{r_-(\mu_r)}\Bigr],\qquad
  r_\mp(\mu)=s_\mp+(1-p_\pm)\frac{1-\mu}{\mu},
\]
and it contains values on both sides of one whenever $r_-(\mu_a)/r_+(\mu_r)<1/\mathrm{AR}<r_+(\mu_a)/r_-(\mu_r)$.
\end{lemma}

\emph{Proof of \cref{thm:misclass}(c) and \cref{lem:mc-range}.} By \cref{lem:id-count}, $q_\pi/\mu_\pi=\Se_\pi+(1-\Sp_\pi)(1-\mu_\pi)/\mu_\pi=r_\pi$, so $q_a/q_r=(r_a\mu_a)/(r_r\mu_r)=\mathrm{AR}\,r_a/r_r$. With pooled rates, the corrected arm means converge to $(q_\pi+\overline\Sp-1)/\overline J$, and $q_\pi+\overline\Sp-1=J_\pi\mu_\pi+\overline\Sp-\Sp_\pi$ by \cref{lem:id-count}. The factor $\overline J$ cancels from the ratio, which gives \eqref{eq:mc-diff}. For the range, fix $\mu\in(0,1)$. The function $(\Se,\Sp)\mapsto\Se+(1-\Sp)(1-\mu)/\mu$ is continuous, increasing in $\Se$ and decreasing in $\Sp$, so on the rectangle $[s_-,s_+]\times[p_-,p_+]$ it takes every value between its minimum $r_-(\mu)$ at $(s_-,p_+)$ and its maximum $r_+(\mu)$ at $(s_+,p_-)$. On this rectangle $J\ge s_-+p_--1>0$. The two arms' rates vary independently, so $r_a/r_r$ ranges over the image of a product of two connected sets under a continuous map, which is the interval with endpoints $r_-(\mu_a)/r_+(\mu_r)$ and $r_+(\mu_a)/r_-(\mu_r)$. Multiplying by $\mathrm{AR}$ gives the stated set, which contains values below and above one exactly when $r_-(\mu_a)/r_+(\mu_r)<1/\mathrm{AR}<r_+(\mu_a)/r_-(\mu_r)$. For the mixture model, the expected number of recorded positions of type $k$ is $kTS\,e_{\pi k}$, and each is flagged with probability $q_k$. Hence $\E\sum\ind{L_c=1,\widehat L_c=1}=kTS\sum_{k:L_k=1}e_{\pi k}q_k$ and $\E\sum\ind{L_c=1}=kTS\,\mu_\pi$, which gives $\Se_\pi$, and $1-\Sp_\pi$ follows in the same way. Thus $\Se_a=\Se_r$ if and only if $\sum_{k:L_k=1}(e_{ak}/\mu_a-e_{rk}/\mu_r)q_k=0$, and similarly for the specificities.

\emph{Proof of \cref{thm:misclass}(d).} If $(\widehat\Se_\pi,\widehat\Sp_\pi)\to(\Se_\pi,\Sp_\pi)$ in probability and $\widehat q_\pi\to q_\pi$, then, since $\mathrm{RG}_\pi$ is continuous where $J_\pi>0$, $\widehat\mu^{\mathrm{st}}_\pi\to(q_\pi+\Sp_\pi-1)/J_\pi=\mu_\pi$ by \cref{lem:id-count}. This uses no relation between the arms. For the identity \eqref{eq:mc-ds}, let arm $\pi$ have $\mathrm{PP}$ flagged and $\mathrm{PN}$ unflagged recorded positions, so that $\widehat q_\pi=\mathrm{PP}/(\mathrm{PP}+\mathrm{PN})$. Let $x_1$ of the $m_1$ coded flagged positions and $x_0$ of the $m_0$ coded unflagged positions be antisemitic, and define the weighted counts $\mathrm{TP}=\mathrm{PP}\,x_1/m_1$, $\mathrm{FP}=\mathrm{PP}\,(m_1-x_1)/m_1$, $\mathrm{FN}=\mathrm{PN}\,x_0/m_0$ and $\mathrm{TN}=\mathrm{PN}\,(m_0-x_0)/m_0$. Then $\mathrm{TP}+\mathrm{FP}=\mathrm{PP}$ and $\mathrm{FN}+\mathrm{TN}=\mathrm{PN}$. Let $\mathrm{AP}=\mathrm{TP}+\mathrm{FN}$ and $\mathrm{TOT}=\mathrm{PP}+\mathrm{PN}$, so that $\widehat\Se_\pi=\mathrm{TP}/\mathrm{AP}$ and $\widehat\Sp_\pi=\mathrm{TN}/(\mathrm{TOT}-\mathrm{AP})$. Then
\[
  \widehat q_\pi+\widehat\Sp_\pi-1=\frac{\mathrm{PP}}{\mathrm{TOT}}-\frac{\mathrm{FP}}{\mathrm{TOT}-\mathrm{AP}}=\frac{\mathrm{TP}(\mathrm{TOT}-\mathrm{AP})-\mathrm{FP}\,\mathrm{AP}}{\mathrm{TOT}(\mathrm{TOT}-\mathrm{AP})},\qquad
  \widehat\Se_\pi+\widehat\Sp_\pi-1=\frac{\mathrm{TP}(\mathrm{TOT}-\mathrm{AP})-\mathrm{FP}\,\mathrm{AP}}{\mathrm{AP}(\mathrm{TOT}-\mathrm{AP})},
\]
using $\mathrm{PP}=\mathrm{TP}+\mathrm{FP}$ in the first numerator. Whenever $\widehat J_\pi\neq0$ the ratio is $\mathrm{AP}/\mathrm{TOT}=\widehat q_\pi(x_1/m_1)+(1-\widehat q_\pi)(x_0/m_0)$, which is \eqref{eq:mc-ds}. Applied to the records of one puppet, with coded positions drawn without replacement within its two strata, the hypergeometric means give $\E(x_1/m_1\mid\mathcal R_u)=\mathrm{TP}_u/\mathrm{PP}_u$ and $\E(x_0/m_0\mid\mathcal R_u)=\mathrm{FN}_u/\mathrm{PN}_u$, so the conditional mean of the right side is $(\mathrm{TP}_u+\mathrm{FN}_u)/(kTS)=\bar E_u$. An empty stratum contributes zero, exactly. With the unflagged stratum split into score bands, the same argument applies band by band.

\emph{Proof of \cref{thm:misclass}(e).} This is the last identity of \cref{lem:id-count}, divided between the arms.
\qed

\subsection{Proof of \texorpdfstring{\cref{thm:sequential}}{the validity of Algorithm 1}}\label{app:pf-sequential}

Let $a>0$ and $f(n)=(n^\ast)^{1/2}h(n/n^\ast)$, so that the radius of \cref{alg:sequential} is $h_n=\widehat\sigma_nf(n)/n$. Since $\log((1+as)^{1/2}/\alpha)\ge\log(1/\alpha)$, also $f(n)^2\ge2\log(1/\alpha)\,n$ for every $n$.

\begin{lemma}[Time-uniform crossing with a plug-in scale]\label{lem:seq-crossing}
Let $\psi_1,\psi_2,\dots$ be independent and identically distributed, bounded, with mean zero and variance $\sigma^2_\psi\in(0,\infty)$. Let $R_n$ and $\widehat\sigma_n$ be random variables on the same space, with $R_n=O(\log\log n)$ and $\widehat\sigma_n\to\sigma_\psi$ almost surely. Fix $c>0$ and let $n_{\mathrm b}=\lceil cn^\ast\rceil$. Then
\[
  \limsup_{n^\ast\to\infty}\Pr\Bigl(\exists\,n\ge n_{\mathrm b}:\Bigl|\sum_{i\le n}\psi_i+R_n\Bigr|>\widehat\sigma_nf(n)\Bigr)\le\bar\alpha(c)=\Pr\bigl(\exists\,s\ge c:|\mathbb B_s|\ge h(s)\bigr)\le\alpha,
\]
where $\mathbb B$ is a standard Brownian motion. Moreover $\bar\alpha(c)\to\alpha$ as $c\downarrow0$.
\end{lemma}

\begin{proof}
\emph{Step 1: Brownian motion.} For $x\in\mathbb R$ let $\mathcal E_s(x)=\exp(x\mathbb B_s-x^2s/2)$. Mixing over $x\sim\mathrm N(0,a)$ gives
\[
  \mathcal M_s=\int\mathcal E_s(x)\,\frac{e^{-x^2/(2a)}}{(2\pi a)^{1/2}}\,dx=(1+as)^{-1/2}\exp\Bigl\{\frac{a\mathbb B_s^2}{2(1+as)}\Bigr\},
\]
a continuous nonnegative martingale with $\mathcal M_0=1$ \citep{robbins1970}. A direct computation shows that $\mathcal M_s\ge1/\alpha$ if and only if $|\mathbb B_s|\ge h(s)$. By Ville's inequality \citep{ville1939}, $\Pr(\sup_s\mathcal M_s\ge1/\alpha)\le\alpha$, so $\bar\alpha(c)\le\bar\alpha(0)\le\alpha$. For the limit $c\downarrow0$: by the law of the iterated logarithm $\mathcal M_s\to0$ almost surely. Let $s^\ast=\inf\{s:\mathcal M_s\ge1/\alpha\}$. The stopped martingale $\mathcal M_{s\wedge s^\ast}$ is bounded by $1/\alpha$ and has mean one, and it converges to $\alpha^{-1}\ind{s^\ast<\infty}$, so bounded convergence gives $\Pr(s^\ast<\infty)=\alpha$, that is, $\bar\alpha(0)=\alpha$. Since $|\mathbb B_0|=0<h(0)$ and the paths are continuous, $\{s^\ast<\infty\}=\bigcup_{c>0}\{\exists\,s\ge c:|\mathbb B_s|\ge h(s)\}$, an increasing union as $c\downarrow0$, so $\bar\alpha(c)\uparrow\alpha$. Finally, let $G_c(y)=\Pr\{\sup_{s\ge c}|\mathbb B_s|/h(s)\ge y\}$. The supremum over $[c,\infty)$ is attained almost surely, because the ratio is continuous and tends to zero, so $G_c(1)=\bar\alpha(c)$, and $G_c(1-\epsilon)\downarrow G_c(1)$ as $\epsilon\downarrow0$ by continuity from above.

\emph{Step 2: coupling.} Because the $\psi_i$ are bounded, they have a finite moment generating function, and the strong approximation of \citet{komlos1975,komlos1976} gives, after enlarging the probability space if necessary, a standard Brownian motion $\mathbb W$ with $\sum_{i\le n}\psi_i-\sigma_\psi\mathbb W_n=O(\log n)$ almost surely. The enlargement can carry the whole data sequence, by drawing each observation from its conditional law given $\psi_i$, so $R_n$ and $\widehat\sigma_n$ live on the same space. Fix $\epsilon\in(0,1/3)$. Since $f(n)\ge\{2\log(1/\alpha)n\}^{1/2}$, the ratio $\{|\sum_{i\le n}\psi_i-\sigma_\psi\mathbb W_n|+|R_n|\}/f(n)$ tends to zero almost surely, and so does $\sup_{n\ge n'}|\widehat\sigma_n-\sigma_\psi|$ as $n'\to\infty$. Hence the event $E_{n^\ast}$ that, for all $n\ge n_{\mathrm b}$, $|\sum_{i\le n}\psi_i-\sigma_\psi\mathbb W_n|+|R_n|\le\epsilon\sigma_\psi f(n)$ and $\widehat\sigma_n\ge(1-\epsilon)\sigma_\psi$, has probability tending to one as $n^\ast\to\infty$. On $E_{n^\ast}$, for $n\ge n_{\mathrm b}$, the event $|\sum_{i\le n}\psi_i+R_n|>\widehat\sigma_nf(n)$ implies $|\mathbb W_n|>(1-2\epsilon)f(n)$. By Brownian scaling, $\mathbb B_s=(n^\ast)^{-1/2}\mathbb W_{sn^\ast}$ is a standard Brownian motion, and $|\mathbb W_n|>(1-2\epsilon)f(n)$ is the event $|\mathbb B_{n/n^\ast}|>(1-2\epsilon)h(n/n^\ast)$ with $n/n^\ast\ge c$. Hence the probability in the lemma is at most $\Pr(E_{n^\ast}^c)+G_c(1-2\epsilon)$. Letting $n^\ast\to\infty$ and then $\epsilon\downarrow0$ gives the bound $G_c(1)=\bar\alpha(c)$.
\end{proof}

Tying the mixture to $n^\ast$ is essential. With a mixture that does not depend on $n^\ast$, $f(n)$ grows like $(n\log n)^{1/2}$, faster than the law of the iterated logarithm allows a random walk to grow, and the probability of a crossing after $n'$ pairs tends to zero as $n'\to\infty$ for every $\alpha$. A limit statement of that kind is true but carries no information about the level.

\emph{Proof of \cref{thm:sequential}(a).} Let $\vartheta=\E O_a/\E O_r=\lambda\mathrm{AR}$. Suppose $\log\mathrm{AR}\notin C_n$. If $\log\mathrm{AR}<\log\widehat\vartheta_n-\log\lambda_+-h_n$, then $\log\widehat\vartheta_n-\log\vartheta>h_n+\log(\lambda_+/\lambda)\ge h_n$. If $\log\mathrm{AR}>\log\widehat\vartheta_n-\log\lambda_-+h_n$, then $\log\widehat\vartheta_n-\log\vartheta<-h_n-\log(\lambda/\lambda_-)\le-h_n$. In both cases $n|\log\widehat\vartheta_n-\log\vartheta|>\widehat\sigma_nf(n)$. Now $O_u\in[0,1]$, so $\psi_i=O_{a,i}/\E O_a-O_{r,i}/\E O_r$ is bounded. A second-order Taylor expansion of the logarithm gives $n(\log\widehat\vartheta_n-\log\vartheta)=\sum_{i\le n}\psi_i+R_n$ with $|R_n|\le Cn\{(\bar O_a-\E O_a)^2+(\bar O_r-\E O_r)^2\}$ for large $n$, and by the law of the iterated logarithm $R_n=O(\log\log n)$ almost surely. Moreover, $\widehat\sigma_n^{\,2}\to\E\psi_1^2=\sigma^2_\psi$ almost surely, by the strong law of large numbers and the continuity of $(x,y)\mapsto(O_a/x-O_r/y)^2$ at $(\E O_a,\E O_r)$, where both are positive. \Cref{lem:seq-crossing} gives $\limsup_{n^\ast\to\infty}\Pr(\exists\,n\ge n_{\mathrm b}:\log\mathrm{AR}\notin C_n)\le\bar\alpha(c)\le\alpha$, with $\bar\alpha(c)\to\alpha$ as $c\to0$. If $N\ge n_{\mathrm b}$, then $\{\log\mathrm{AR}\notin C_N\}\subseteq\{\exists\,n\ge n_{\mathrm b}:\log\mathrm{AR}\notin C_n\}$. Finally, rule S1 states that $\log\mathrm{AR}>\log\delta$ when $\mathrm{LB}>\log\delta$. If the truth is $\log\mathrm{AR}\le\log\delta$, then $\log\mathrm{AR}<\mathrm{LB}$, so some $C_n$ after the burn-in excludes $\log\mathrm{AR}$, because $\mathrm{LB}$ and $\mathrm{UB}$ are formed only from sets after the burn-in. The same argument applies to S2.

\emph{Proof of \cref{thm:sequential}(b).} Let $\vartheta^\ast=\E O_a/\E O_r$ and $X_i=O_{a,i}-\vartheta^\ast O_{r,i}\in[-\vartheta^\ast,1]$. The $X_i$ are independent with mean zero. Let $b^+_i\in[0,c/\vartheta^\ast]$ and $b^-_i\in[0,c]$, with $c<1$, be predictable bets, that is, functions of $X_1,\dots,X_{i-1}$. Then $1+b^+_iX_i\ge1-c>0$ and $1-b^-_iX_i\ge1-c>0$, and $\E(1\pm b^\pm_iX_i\mid X_1,\dots,X_{i-1})=1$. Hence $\mathrm{Cap}^\pm_n=\prod_{i\le n}(1\pm b^\pm_iX_i)$ are nonnegative martingales with initial value one, and so is $\mathrm{Cap}_n=\frac12\mathrm{Cap}^+_n+\frac12\mathrm{Cap}^-_n$, the hedged capital process of \citet{waudbysmith2024betting}. By Ville's inequality, $\Pr(\exists\,n:\mathrm{Cap}_n(\vartheta^\ast)\ge1/\alpha)\le\alpha$. On the complementary event, $\vartheta^\ast$ is never rejected, and since $\lambda\in[\lambda_-,\lambda_+]$, $\log\mathrm{AR}=\log\vartheta^\ast-\log\lambda\in C^{\mathrm{bet}}_n$ for every $n$.
\qed

\subsection{Proof of \texorpdfstring{\cref{prop:power}}{the power proposition}}\label{app:pf-power}

\emph{Variances.} Assume, as in \eqref{eq:pw-var}, that given the true labels $\mathcal L_u$ of a puppet's recorded positions the flags are independent, with probability $\Se$ for a position with $L_c=1$ and $1-\Sp$ for one with $L_c=0$. Then $\E(\bar Q_u\mid\mathcal L_u)=(1-\Sp)+J\bar E_u$ and $\Var(\bar Q_u\mid\mathcal L_u)=(kTS)^{-1}\{\bar E_u\Se(1-\Se)+(1-\bar E_u)\Sp(1-\Sp)\}$. The law of total variance gives $\sigma^2_\pi=J^2\Var(\bar E_u)+\sigma^2_{\mathrm c,\pi}$, and $\Var(\bar E_u)=\sigma^2_{E,\pi}$ by the computation in the proof of \cref{thm:identify}(b), applied to the true session exposures.

\emph{(a)} By \cref{prop:mc-variance}, with $n$ puppets per arm, $\log\widehat{\mathrm{AR}}$ is approximately normal with mean $\log r_1$ and variance $J^{-2}\sum_\pi\sigma^2_\pi/(n\mu_\pi^2)+V_{\mathrm{val}}=V/n+V_{\mathrm{val}}$. For the pooled design, $V_{\mathrm{val}}=V_{\mathrm{val}}^{\mathrm{pool}}$ is the last term of \eqref{eq:mc-varAR}. For the design stratified by arm, the gradient of $\log\mu_\pi=\log(q_\pi+\Sp_\pi-1)-\log J_\pi$ with respect to $(q_\pi,\Se_\pi,\Sp_\pi)$ is $J_\pi^{-1}\{1/\mu_\pi,-1,(1-\mu_\pi)/\mu_\pi\}$, and independence of the validation samples across arms and labels gives $V_{\mathrm{val}}^{\mathrm{st}}$. For double sampling, $\log\mu_\pi=\log\{q_\pi\PPV_\pi+(1-q_\pi)\FOR_\pi\}$ has gradient $\mu_\pi^{-1}(\PPV_\pi-\FOR_\pi,q_\pi,1-q_\pi)$ with respect to $(q_\pi,\PPV_\pi,\FOR_\pi)$, and independent coding within the two strata gives $V_{\mathrm{val}}^{\mathrm{ds}}$ and the factor $(\PPV_\pi-\FOR_\pi)^2$ in place of $J_\pi^{-2}$. Take $r_1>r_0$; the other case is symmetric. The interval excludes $\log r_0$ from above with probability $\Phi\{\Delta(V/n+V_{\mathrm{val}})^{-1/2}-z_{1-\alpha/2}\}$, and the probability of excluding it from below is negligible. This is at least $1-\beta$ if and only if $V/n+V_{\mathrm{val}}\le\Delta^2/z^2$. If $\Delta^2\le z^2V_{\mathrm{val}}$ this is impossible, and otherwise it is equivalent to $n\ge z^2V/(\Delta^2-z^2V_{\mathrm{val}})$. Letting the other validation stratum grow without bound and setting the remaining sizes equal across arms, the condition $\Delta^2>z^2V_{\mathrm{val}}$ becomes $n_0>n_0^{\min}$, $n_{0\pi}>n^{\min}_{0,\mathrm{st}}$ or $m_{0\pi}>m_0^{\min}$ for the three designs. With a perfect classifier, $J=1$ and $\sigma^2_{\mathrm c,\pi}=0$, so $V=V_0$.

\emph{(b)} Expanding \eqref{eq:pw-var}, $\sigma^2_{E,\pi}+\sigma^2_{\mathrm c,\pi}/J^2=\sigma^2_{\mathrm s,\pi}\gamma_\pi+\{\sigma^2_{\mathrm s,\pi}(1-\gamma_\pi)+\varsigma^2_\pi/(kTJ^2)\}/S$, so $V=v_{\mathrm p}+v_{\mathrm s}/S$. With $V_{\mathrm{val}}=0$, the required $n$ is proportional to $V$, and the cost is proportional to $g(S)=(v_{\mathrm p}+v_{\mathrm s}/S)(c_p+Sc_s)=v_{\mathrm p}c_p+v_{\mathrm s}c_s+v_{\mathrm p}c_sS+v_{\mathrm s}c_p/S$. This is convex on $(0,\infty)$, and $g'(S)=v_{\mathrm p}c_s-v_{\mathrm s}c_p/S^2$ vanishes at $S^\ast$.

\emph{(c)} The radius per unit standard deviation at $n^\ast$ is $f(n^\ast)/n^\ast=h(1)(n^\ast)^{-1/2}=c_\alpha(n^\ast)^{-1/2}$. With known nuisance, the standard deviation per pair of the linearised log ratio is $V^{1/2}$, so the radius at $n^\ast$ is $c_\alpha(V/n^\ast)^{1/2}$. Under the same normal approximation as in (a), the lower end of $C_n$ exceeds $\log r_0$ with probability $\Phi\{\Delta(n/V)^{1/2}-c_\alpha\}$, which is at least $1-\beta$ when $n\ge n_{\mathrm{seq}}$. The running lower end $\mathrm{LB}$ of \cref{alg:sequential} is at least the lower end of $C_n$, so the power by the horizon is at least the power at the horizon.

\emph{(d)} By \cref{lem:seq-unbiased}, $\Var(Y_u)=\Var\{\mathrm{TP}_u/(kTS)\}+\E\Var(Y_u\mid\mathcal R_u)$. Under the flag model above, $\E\{\mathrm{TP}_u/(kTS)\mid\mathcal L_u\}=\Se\bar E_u$ and $\Var\{\mathrm{TP}_u/(kTS)\mid\mathcal L_u\}=\bar E_u\Se(1-\Se)/(kTS)$, so $\Var\{\mathrm{TP}_u/(kTS)\}=\Se^2\sigma^2_{E,\pi}+\mu_\pi\Se(1-\Se)/(kTS)$. Given the record, $Y_u$ is $\mathrm{PP}_u/(kTS)$ times a hypergeometric proportion, with variance $\{\mathrm{PP}_u/(kTS)\}^2\PPV_u(1-\PPV_u)m^{-1}(\mathrm{PP}_u-m)/(\mathrm{PP}_u-1)$ when $\mathrm{PP}_u>m$, where $\PPV_u=\mathrm{TP}_u/\mathrm{PP}_u$. Replacing $\mathrm{PP}_u/(kTS)$ by $q_\pi$ and $\PPV_u$ by $\PPV_\pi$, and dropping the finite-population factor, gives $q_\pi^2\PPV_\pi(1-\PPV_\pi)/m$. Dividing by $(\E Y_u)^2=(\Se\mu_\pi)^2$ and summing over the arms gives $V_Y$. Finally, the lower end of $C_n$ is $\log\widehat\vartheta_n-\log\lambda_+-h_n$, and $\log\vartheta=\log r_1+\log\lambda_S\ge\log r_1+\log\lambda_-$. In the worst case $\lambda_S=\lambda_-$, the lower end is centred at $\log r_1-\log(\lambda_+/\lambda_-)$, and the argument of (c) with $\Delta$ replaced by $\Delta-\log(\lambda_+/\lambda_-)$ gives the horizon.
\qed

\begin{remark}[A model-based session variance]\label{rem:pw-markov}
If sessions follow the chain of \cref{asm:markov} with $k=1$ and start near stationarity, \cref{prop:markov}(a) gives $\sigma^2_{\mathcal A}/T$ for the within-session part of $\sigma^2_{\mathrm s,\pi}$. By the law of total variance, this is approximately a lower bound for $\sigma^2_{\mathrm s,\pi}$, since it ignores differences between puppets and between sessions. It is a check on pilot estimates, not a substitute for them.
\end{remark}
